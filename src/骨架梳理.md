# 数据结构
```C
/* ============ 全局状态 ============ */

static volatile sig_atomic_t g_running = 1;

static struct rte_ring    *g_rx_ring    = NULL;
static struct rte_mempool *g_desc_pool  = NULL;
static struct rte_mempool *g_cpbuf_pool = NULL; //缓冲区、存在拷贝

static int g_sockfd = -1; //UDP socket文件描述符

/* ============ 数据结构 ============ */

struct rxbuf_desc {
    uint8_t  *data;
    uint32_t  len;
    uint8_t  *cpbuf;   /* mempool 槽位；归还 g_cpbuf_pool */
};

struct writer_ctx {
    const char             *output_path;
    int                     writer_cpu;
    volatile int            state;     /* 0=init, 1=ready, -1=failed */
    volatile int            rx_done;
    uint64_t                writer_errors;
    struct udp_writer_stats wstats;
};

struct app_config { //应用层运行参数，收包与写盘行为的静态配置 
    const char *output_path;
    uint16_t    dst_port;
    uint32_t    expected_pkts;
    int         raw_payload;
    uint32_t    expected_payload_size;
    int         rx_cpu;
    int         writer_cpu;
};

struct rx_stats { //RX收包统计信息
    uint64_t total_rx;
    uint64_t total_rx_bytes;
    uint64_t udp_matched;
    uint64_t udp_bytes;
    uint64_t payload_errors;
    uint64_t writer_errors;
    uint64_t buf_drops;
    uint64_t oversize_drops;
    uint64_t ring_full_drops;
    uint64_t bufpool_peak;
    struct udp_payload_validate_stats payload_validate;
};

struct rx_loop_ctx { //RX主循环上下文,聚合所需信息
    const struct app_config      *cfg;
    struct rx_stats              *stats;
    struct udp_rx_burst_timing   *burst;
    struct writer_ctx            *wctx;
    uint32_t                      next_expected_seq;
    uint32_t                      raw_seq;
    uint64_t                      tsc_start_ns;
    int                           timing_active;
};
```
